    <script>
        const ctx = document.getElementById('lucroChart').getContext('2d');
        let chart;

        function switchTab(tab) {
            document.getElementById('view-pl').classList.toggle('hidden', tab !== 'pl');
            document.getElementById('view-tax').classList.toggle('hidden', tab !== 'tax');
            document.getElementById('view-cash').classList.toggle('hidden', tab !== 'cash');
            
            document.getElementById('tab-pl').classList.toggle('active', tab === 'pl');
            document.getElementById('tab-tax').classList.toggle('active', tab === 'tax');
            document.getElementById('tab-cash').classList.toggle('active', tab === 'cash');
        }

        function updateChart(dataValues) {
            const data = {
                labels: ['Food', 'Payroll', 'Gov', 'Rent', 'Variable', 'Profit'],
                datasets: [{
                    data: dataValues,
                    backgroundColor: ['#f87171', '#60a5fa', '#fbbf24', '#818cf8', '#f472b6', '#34d399'],
                    borderWidth: 0
                }]
            };
            if (chart) chart.destroy();
            chart = new Chart(ctx, {
                type: 'doughnut',
                data: data,
                options: { responsive: true, maintainAspectRatio: false, cutout: '70%', plugins: { legend: { position: 'bottom', labels: { boxWidth: 8, font: { size: 10 } } } } }
            });
        }

        function calculate(isManualInput = false) {
            const vendas = parseFloat(document.getElementById('vendas').value) || 0;
            
            if (!isManualInput) {
                document.getElementById('cogs_val').value = (vendas * 0.33).toFixed(0);
                document.getElementById('uber_val').value = (vendas * 0.10).toFixed(0);
                document.getElementById('card_val').value = (vendas * 0.025).toFixed(0);
            }

            const cogs = parseFloat(document.getElementById('cogs_val').value) || 0;
            const labor = parseFloat(document.getElementById('labor_val').value) || 0;
            const uber = parseFloat(document.getElementById('uber_val').value) || 0;
            const card = parseFloat(document.getElementById('card_val').value) || 0;
            const rent = parseFloat(document.getElementById('rent_val').value) || 0;
            const utils = parseFloat(document.getElementById('utils_val').value) || 0;

            // --- TAXES ---
            const gstCollected = vendas * 0.05;
            const qstCollected = vendas * 0.09975;
            const taxCollectedTotal = gstCollected + qstCollected;
            
            const creditBase = rent + utils + uber; 
            const itc = creditBase * 0.05;
            const itr = creditBase * 0.09975;
            const taxCreditsTotal = itc + itr;
            
            const taxRemittance = taxCollectedTotal - taxCreditsTotal;

            // --- P&L ---
            const das = labor * 0.135; 
            const totalOp = cogs + labor + das + uber + card + rent + utils;
            const ebitda = vendas - totalOp;
            const corpTax = ebitda > 0 ? ebitda * 0.122 : 0;
            const netProfit = ebitda - corpTax;

            // --- CASH FLOW LOGIC ---
            const cashIn = vendas + taxCollectedTotal;
            const cashOutCogs = cogs; // Food is 0%
            const cashOutLabor = labor + das;
            const cashOutOps = (rent + utils + uber) * 1.14975 + card;
            const operationalBalance = cashIn - cashOutCogs - cashOutLabor - cashOutOps;
            const freeCashFlow = operationalBalance - taxRemittance - corpTax;

            // --- UI UPDATE ---
            document.getElementById('final_profit').innerText = '$' + Math.max(0, netProfit).toLocaleString(undefined, {minimumFractionDigits: 2});
            const pPerc = vendas > 0 ? (netProfit / vendas * 100).toFixed(1) : 0;
            document.getElementById('profit_perc_badge').innerText = pPerc + '%';
            
            document.getElementById('tax_collected_total').innerText = '$' + taxCollectedTotal.toLocaleString();
            document.getElementById('tax_credits_total').innerText = '$' + taxCreditsTotal.toLocaleString();
            document.getElementById('tax_remittance').innerText = '$' + Math.max(0, taxRemittance).toLocaleString(undefined, {minimumFractionDigits: 2});

            // Cash Flow Table
            document.getElementById('cash_vendas_in').innerText = '$' + cashIn.toLocaleString();
            document.getElementById('cash_vendas_net').innerText = '$' + cashIn.toLocaleString();
            document.getElementById('cash_cogs_out').innerText = '$' + cashOutCogs.toLocaleString();
            document.getElementById('cash_cogs_net').innerText = '-$' + cashOutCogs.toLocaleString();
            document.getElementById('cash_labor_out').innerText = '$' + cashOutLabor.toLocaleString();
            document.getElementById('cash_labor_net').innerText = '-$' + cashOutLabor.toLocaleString();
            document.getElementById('cash_ops_out').innerText = '$' + cashOutOps.toLocaleString();
            document.getElementById('cash_ops_net').innerText = '-$' + cashOutOps.toLocaleString();
            document.getElementById('cash_bank_bal').innerText = '$' + operationalBalance.toLocaleString();
            document.getElementById('cash_tax_out').innerText = '$' + (taxRemittance + corpTax).toLocaleString();
            document.getElementById('cash_tax_net').innerText = '-$' + (taxRemittance + corpTax).toLocaleString();
            document.getElementById('cash_final_net').innerText = '$' + freeCashFlow.toLocaleString(undefined, {minimumFractionDigits: 2});

            updateChart([cogs, labor, das + corpTax, rent + utils, uber + card, Math.max(0, netProfit)]);

            // Alerts
            const alerts = document.getElementById('alerts_container');
            alerts.innerHTML = '';
            if (vendas > 0 && freeCashFlow < 0) {
                alerts.innerHTML += `<div class="p-3 bg-red-100 border-l-4 border-red-500 text-red-700 text-[11px] font-bold">⚠️ CRITICAL: Negative Free Cash Flow. You are spending your tax reserve!</div>`;
            }
        }

        document.getElementById('vendas').addEventListener('input', () => calculate(false));
        document.querySelectorAll('input').forEach(input => {
            if(input.id !== 'vendas') input.addEventListener('input', () => calculate(true));
        });

        calculate();
    </script>
